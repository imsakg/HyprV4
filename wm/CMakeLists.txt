add_library(hyprv4_core INTERFACE)
target_include_directories(hyprv4_core INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(hyprv4_core INTERFACE cxx_std_17)
if(BUILD_TESTING)
  file(GLOB test_sources CONFIGURE_DEPENDS tests/test_*.cpp)
  foreach(source IN LISTS test_sources)
    get_filename_component(name ${source} NAME_WE)
    add_executable(${name} ${source})
    target_link_libraries(${name} PRIVATE hyprv4_core)
    if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
      target_compile_options(${name} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
    add_test(NAME ${name} COMMAND ${name})
  endforeach()
endif()
